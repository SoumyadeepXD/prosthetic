/**
 * ==============================================================================
 * VAPA -- Biomimetic Prosthetic System: ESP32 Multi-Sensor & Supervisor Firmware
 * ==============================================================================
 * Single-file standalone firmware for Arduino IDE or PlatformIO.
 *
 * SENSORS & PERIPHERALS:
 *   - EMG A0/A1 & EEG A2: ADS1115 (0x48) via I2C (SDA=GPIO21, SCL=GPIO22)
 *   - 5x FSR402: GPIO 32 (Thumb), 33 (Index), 34 (Middle), 35 (Ring), 36 (Little)
 *   - E-Stop Switch: GPIO 27 (Active LOW, INPUT_PULLUP)
 *   - PCA9685 Servo OE: GPIO 25 (Active LOW, fail-closed; HIGH on boot)
 *   - High-Speed UART2 to Jetson: TX=GPIO17, RX=GPIO16 @ 460800 baud (Binary protocol)
 *   - USB Serial Monitor (to Laptop): 115200 baud diagnostics
 *
 * NOTE: TCA9548A multiplexer and AS5600 encoders have been fully removed.
 * ==============================================================================
 */

static inline uint16_t crc16_ccitt(const uint8_t *data, size_t len) {
    uint16_t crc = 0xFFFF;
    for (size_t i = 0; i < len; ++i) {
        crc = (uint16_t)(((crc << 8) & 0xFFFF) ^ crc16_table[((crc >> 8) ^ data[i]) & 0xFF]);
    }
    return crc;
}

static size_t frame_encode(uint8_t type, uint8_t seq, const uint8_t *payload, uint8_t p_len, uint8_t *out_frame, size_t max_len) {
    size_t total = VAPA_HEADER_SIZE + p_len + VAPA_FOOTER_SIZE;
    if (total > max_len || !out_frame) return 0;

    out_frame[0] = VAPA_SYNC0;
    out_frame[1] = VAPA_SYNC1;
    out_frame[2] = p_len;
    out_frame[3] = type;
    out_frame[4] = seq;

    if (payload && p_len > 0) {
        memcpy(&out_frame[5], payload, p_len);
    }

    uint16_t crc = crc16_ccitt(&out_frame[2], 3 + p_len);
    out_frame[5 + p_len] = (uint8_t)(crc & 0xFF);
    out_frame[6 + p_len] = (uint8_t)((crc >> 8) & 0xFF);
    return total;
}

// Streaming Parser
enum parser_state_t {
    P_WAIT_SYNC0 = 0,
    P_WAIT_SYNC1,
    P_WAIT_LEN,
    P_WAIT_TYPE,
    P_WAIT_SEQ,
    P_WAIT_PAYLOAD,
    P_WAIT_CRC_L,
    P_WAIT_CRC_H
};

struct frame_parser_t {
    parser_state_t state;
    uint8_t payload_len;
    uint8_t msg_type;
    uint8_t seq;
    uint8_t payload_idx;
    uint8_t payload_buf[256];
    uint16_t crc_received;
    uint16_t crc_errors;

    void init() {
        memset(this, 0, sizeof(frame_parser_t));
        state = P_WAIT_SYNC0;
    }

    bool feed(uint8_t b, uint8_t *out_type, uint8_t *out_seq, const uint8_t **out_payload, uint8_t *out_len) {
        switch (state) {
            case P_WAIT_SYNC0:
                if (b == VAPA_SYNC0) state = P_WAIT_SYNC1;
                break;
            case P_WAIT_SYNC1:
                if (b == VAPA_SYNC1) state = P_WAIT_LEN;
                else if (b == VAPA_SYNC0) state = P_WAIT_SYNC1;
                else state = P_WAIT_SYNC0;
                break;
            case P_WAIT_LEN:
                payload_len = b;
                state = P_WAIT_TYPE;
                break;
            case P_WAIT_TYPE:
                msg_type = b;
                state = P_WAIT_SEQ;
                break;
            case P_WAIT_SEQ:
                seq = b;
                payload_idx = 0;
                state = (payload_len == 0) ? P_WAIT_CRC_L : P_WAIT_PAYLOAD;
                break;
            case P_WAIT_PAYLOAD:
                payload_buf[payload_idx++] = b;
                if (payload_idx >= payload_len) state = P_WAIT_CRC_L;
                break;
            case P_WAIT_CRC_L:
                crc_received = (uint16_t)b;
                state = P_WAIT_CRC_H;
                break;
            case P_WAIT_CRC_H: {
                crc_received |= ((uint16_t)b << 8);
                state = P_WAIT_SYNC0;

                uint8_t check_buf[3 + 256];
                check_buf[0] = payload_len;
                check_buf[1] = msg_type;
                check_buf[2] = seq;
                if (payload_len > 0) {
                    memcpy(&check_buf[3], payload_buf, payload_len);
                }
                uint16_t calc = crc16_ccitt(check_buf, 3 + payload_len);
                if (calc == crc_received) {
                    if (out_type) *out_type = msg_type;
                    if (out_seq) *out_seq = seq;
                    if (out_payload) *out_payload = payload_buf;
                    if (out_len) *out_len = payload_len;
                    return true;
                } else {
                    crc_errors++;
                    return false;
                }
            }
        }
        return false;
    }
};

// ==============================================================================
// 3. SAFETY SUPERVISOR STATE MACHINE
// ==============================================================================
struct supervisor_config_t {
    uint16_t force_ceiling;
    uint16_t hb_timeout_ms;
    uint16_t force_hold_ms;
};

struct supervisor_state_t {
    bool estop_pressed;
    bool oe_enabled;
    bool armed;
    bool heartbeat_ok;
    bool force_fault;
    bool ads_fault;
    bool latched_fault;
    bool had_first_heartbeat;
    bool arm_request_transitioned;
    uint8_t arm_request;
    uint32_t last_heartbeat_time_ms;
    uint32_t force_fault_start_ms;

    void init() {
        memset(this, 0, sizeof(supervisor_state_t));
    }

    void process_heartbeat(uint8_t arm_req, uint32_t now) {
        last_heartbeat_time_ms = now;
        if (!had_first_heartbeat) {
            had_first_heartbeat = true;
            arm_request_transitioned = false; // Require fresh 0 -> 1 transition
        } else {
            if (arm_request == 0 && arm_req == 1) {
                arm_request_transitioned = true;
            } else if (arm_req == 0) {
                arm_request_transitioned = false;
            }
        }
        arm_request = arm_req;
    }

    void update(const supervisor_config_t *cfg, uint32_t now, const uint16_t fsr[5], bool ads_ok) {
        // 1. Heartbeat check
        if (had_first_heartbeat) {
            if ((now - last_heartbeat_time_ms) > cfg->hb_timeout_ms) {
                heartbeat_ok = false;
                latched_fault = true; // Latched on timeout
            } else {
                heartbeat_ok = true;
            }
        } else {
            heartbeat_ok = false;
        }

        // 2. E-Stop check
        if (estop_pressed) {
            latched_fault = true;
        }

        // 3. ADS check
        ads_fault = !ads_ok;

        // 4. Force ceiling
        bool over = false;
        for (int i = 0; i < 5; ++i) {
            if (fsr[i] >= cfg->force_ceiling) { over = true; break; }
        }
        if (over) {
            if (force_fault_start_ms == 0) {
                force_fault_start_ms = now;
            } else {
                uint32_t dur = now - force_fault_start_ms;
                if (dur >= cfg->force_hold_ms) force_fault = true;
                if (dur >= 500) latched_fault = true;
            }
        } else {
            force_fault_start_ms = 0;
            force_fault = false;
        }

        // 5. Output Enable (OE) Evaluation
        if (arm_request == 1 &&
            arm_request_transitioned &&
            heartbeat_ok &&
            !estop_pressed &&
            !latched_fault &&
            !ads_fault &&
            !force_fault) {
            oe_enabled = true;
            armed = true;
        } else {
            oe_enabled = false;
            armed = false;
        }
    }

    bool clear_fault() {
        if (!estop_pressed && arm_request == 0) {
            latched_fault = false;
            force_fault = false;
            force_fault_start_ms = 0;
            arm_request_transitioned = false;
            return true;
        }
        return false;
    }

    uint16_t get_flags() const {
        uint16_t flags = 0;
        if (estop_pressed) flags |= FLAG_ESTOP_PRESSED;
        if (oe_enabled)    flags |= FLAG_OE_ENABLED;
        if (armed)         flags |= FLAG_ARMED;
        if (heartbeat_ok)  flags |= FLAG_HEARTBEAT_OK;
        if (force_fault)   flags |= FLAG_FORCE_FAULT;
        if (ads_fault)     flags |= FLAG_ADS_FAULT;
        if (latched_fault) flags |= FLAG_LATCHED_FAULT;
        return flags;
    }
};

static SemaphoreHandle_t g_data_mutex = NULL;
static uint16_t g_shared_emg[2] = {0, 0};
static uint16_t g_shared_eeg = 0;
static uint16_t g_shared_fsr[5] = {0, 0, 0, 0, 0};
static bool     g_shared_ads_ok = false;
static uint16_t g_ads_err_count = 0;
static uint8_t  g_tx_seq = 0;

static void i2c_bus_recover() {
    Wire.end();
    pinMode(PIN_I2C_SDA, INPUT_PULLUP);
    pinMode(PIN_I2C_SCL, OUTPUT);
    digitalWrite(PIN_I2C_SCL, HIGH);
    delayMicroseconds(5);
    for (int i = 0; i < 9; i++) {
        digitalWrite(PIN_I2C_SCL, LOW);
        delayMicroseconds(5);
        digitalWrite(PIN_I2C_SCL, HIGH);
        delayMicroseconds(5);
    }
    pinMode(PIN_I2C_SDA, OUTPUT);
    digitalWrite(PIN_I2C_SDA, LOW);
    delayMicroseconds(5);
    digitalWrite(PIN_I2C_SCL, HIGH);
    delayMicroseconds(5);
    digitalWrite(PIN_I2C_SDA, HIGH);
    delayMicroseconds(5);
    Wire.begin(PIN_I2C_SDA, PIN_I2C_SCL, 400000);
}

static bool ads1115_start_conversion(uint8_t channel) {
    if (channel > 3) return false;
    uint16_t mux_bits = (0x04 | channel) << 12;
    uint16_t config = 0x8000 | mux_bits | 0x0200 | 0x0100 | 0x00E0 | 0x0003;
    Wire.beginTransmission(ADS1115_ADDR);
    Wire.write(ADS_REG_CFG);
    Wire.write((uint8_t)(config >> 8));
    Wire.write((uint8_t)(config & 0xFF));
    return (Wire.endTransmission() == 0);
}

static bool ads1115_read_conversion(uint16_t *out_val) {
    Wire.beginTransmission(ADS1115_ADDR);
    Wire.write(ADS_REG_CONV);
    if (Wire.endTransmission() != 0) return false;
    if (Wire.requestFrom((uint8_t)ADS1115_ADDR, (uint8_t)2) != 2) return false;
    uint16_t raw = (Wire.read() << 8) | Wire.read();
    if ((int16_t)raw < 0) raw = 0;
    *out_val = raw;
    return true;
}

static inline uint16_t median_of_3(uint16_t a, uint16_t b, uint16_t c) {
    if ((a <= b && b <= c) || (c <= b && b <= a)) return b;
    if ((b <= a && a <= c) || (c <= a && a <= b)) return a;
    return c;
}

void SensorSamplingTask(void *pvParameters) {
    esp_task_wdt_add(NULL);

    static uint16_t fsr_history[5][3] = {{0}};
    static float fsr_ema[5] = {0};
    uint8_t sample_idx = 0;

    uint8_t ads_channel = 0;
    uint16_t emg[2] = {0, 0};
    uint16_t eeg = 0;
    bool ads_ok = true;
    uint8_t consecutive_errors = 0;

    for (;;) {
        esp_task_wdt_reset();

        // 1. Read FSRs (ADC1: GPIO 32-36)
        uint16_t raw_fsr[5];
        raw_fsr[0] = analogRead(PIN_FSR_THUMB);
        raw_fsr[1] = analogRead(PIN_FSR_INDEX);
        raw_fsr[2] = analogRead(PIN_FSR_MID);
        raw_fsr[3] = analogRead(PIN_FSR_RING);
        raw_fsr[4] = analogRead(PIN_FSR_LITTLE);

        uint16_t filtered_fsr[5];
        for (int i = 0; i < 5; ++i) {
            fsr_history[i][sample_idx] = raw_fsr[i];
            uint16_t med = median_of_3(fsr_history[i][0], fsr_history[i][1], fsr_history[i][2]);
            fsr_ema[i] = 0.6f * med + 0.4f * fsr_ema[i];
            filtered_fsr[i] = (uint16_t)fsr_ema[i];
        }
        sample_idx = (sample_idx + 1) % 3;

        // 2. Non-blocking ADS1115 read & cycle
        uint16_t conv_val = 0;
        if (ads1115_read_conversion(&conv_val)) {
            consecutive_errors = 0;
            ads_ok = true;
            if (ads_channel == 0) emg[0] = conv_val;
            else if (ads_channel == 1) emg[1] = conv_val;
            else if (ads_channel == 2) eeg = conv_val;
        } else {
            consecutive_errors++;
            g_ads_err_count++;
            if (consecutive_errors >= 3) {
                ads_ok = false;
                set_hardware_oe(false); // Fail closed immediately
                i2c_bus_recover();
            }
        }

        ads_channel = (ads_channel + 1) % 3;
        if (!ads1115_start_conversion(ads_channel)) {
            consecutive_errors++;
            g_ads_err_count++;
            if (consecutive_errors >= 3) {
                ads_ok = false;
                set_hardware_oe(false);
            }
        }

        // 3. Check physical E-stop immediately
        if (digitalRead(PIN_ESTOP) == LOW) {
            set_hardware_oe(false);
        }

        // 4. Update shared data block
        if (xSemaphoreTake(g_data_mutex, pdMS_TO_TICKS(2)) == pdTRUE) {
            g_shared_emg[0] = emg[0];
            g_shared_emg[1] = emg[1];
            g_shared_eeg = eeg;
            for (int i = 0; i < 5; ++i) g_shared_fsr[i] = filtered_fsr[i];
            g_shared_ads_ok = ads_ok;
            xSemaphoreGive(g_data_mutex);
        }

        vTaskDelay(pdMS_TO_TICKS(2));
    }
}

void SupervisorUartTask(void *pvParameters) {
    esp_task_wdt_add(NULL);

    frame_parser_t parser;
    parser.init();

    uint32_t last_sensor_tx_ms = 0;
    uint32_t last_status_tx_ms = 0;
    uint32_t last_usb_debug_ms = 0;
    uint8_t tx_buf[64];

    for (;;) {
        esp_task_wdt_reset();
        uint32_t now = millis();

        // 1. Process UART2 Incoming from Jetson
        while (Serial2.available() > 0) {
            uint8_t b = Serial2.read();
            uint8_t msg_type = 0;
            uint8_t seq = 0;
            const uint8_t *payload = NULL;
            uint8_t len = 0;

            if (parser.feed(b, &msg_type, &seq, &payload, &len)) {
                if (msg_type == MSG_TYPE_HEARTBEAT && len == 2) {
                    const heartbeat_payload_t *hb = (const heartbeat_payload_t *)payload;
                    g_supervisor.process_heartbeat(hb->arm_request, now);
                } else if (msg_type == MSG_TYPE_CONFIG && len == 6) {
                    const config_payload_t *cfg = (const config_payload_t *)payload;
                    g_config.force_ceiling = cfg->force_ceiling;
                    g_config.hb_timeout_ms = cfg->hb_timeout_ms;
                    g_config.force_hold_ms = cfg->force_hold_ms;
                    ack_nack_payload_t ack = {.ref_type = MSG_TYPE_CONFIG, .ref_seq = seq, .code = 0};
                    size_t enc_len = frame_encode(MSG_TYPE_ACK_NACK, g_tx_seq++, (const uint8_t *)&ack, sizeof(ack), tx_buf, sizeof(tx_buf));
                    Serial2.write(tx_buf, enc_len);
                } else if (msg_type == MSG_TYPE_CLEAR_FAULT && len == 0) {
                    bool cleared = g_supervisor.clear_fault();
                    ack_nack_payload_t ack = {
                        .ref_type = MSG_TYPE_CLEAR_FAULT,
                        .ref_seq = seq,
                        .code = (uint8_t)(cleared ? 0 : 1)
                    };
                    size_t enc_len = frame_encode(MSG_TYPE_ACK_NACK, g_tx_seq++, (const uint8_t *)&ack, sizeof(ack), tx_buf, sizeof(tx_buf));
                    Serial2.write(tx_buf, enc_len);
                }
            }
        }

        // 2. Read E-Stop switch
        g_supervisor.estop_pressed = (digitalRead(PIN_ESTOP) == LOW);
        if (g_supervisor.estop_pressed) {
            set_hardware_oe(false);
        }

        // 3. Snapshot shared sensor data
        uint16_t local_emg[2] = {0};
        uint16_t local_eeg = 0;
        uint16_t local_fsr[5] = {0};
        bool local_ads_ok = false;

        if (xSemaphoreTake(g_data_mutex, pdMS_TO_TICKS(2)) == pdTRUE) {
            local_emg[0] = g_shared_emg[0];
            local_emg[1] = g_shared_emg[1];
            local_eeg = g_shared_eeg;
            for (int i = 0; i < 5; ++i) local_fsr[i] = g_shared_fsr[i];
            local_ads_ok = g_shared_ads_ok;
            xSemaphoreGive(g_data_mutex);
        }

        // 4. Update Supervisor State Machine & OE
        g_supervisor.update(&g_config, now, local_fsr, local_ads_ok);
        set_hardware_oe(g_supervisor.oe_enabled);

        // 5. Send 100 Hz SENSOR packet to Jetson
        if (now - last_sensor_tx_ms >= SENSOR_INTERVAL_MS) {
            last_sensor_tx_ms = now;
            sensor_payload_t s_msg;
            s_msg.t_ms = now;
            s_msg.emg[0] = local_emg[0];
            s_msg.emg[1] = local_emg[1];
            s_msg.eeg = local_eeg;
            for (int i = 0; i < 5; ++i) s_msg.fsr[i] = local_fsr[i];
            s_msg.flags = g_supervisor.get_flags();

            size_t enc_len = frame_encode(MSG_TYPE_SENSOR, g_tx_seq++, (const uint8_t *)&s_msg, sizeof(s_msg), tx_buf, sizeof(tx_buf));
            if (enc_len > 0) Serial2.write(tx_buf, enc_len);
        }

        // 6. Send 2 Hz STATUS packet to Jetson
        if (now - last_status_tx_ms >= STATUS_INTERVAL_MS) {
            last_status_tx_ms = now;
            uint32_t hb_age = (g_supervisor.last_heartbeat_time_ms > 0) ? (now - g_supervisor.last_heartbeat_time_ms) : 65535;
            if (hb_age > 65535) hb_age = 65535;

            status_payload_t st_msg;
            st_msg.fw_version = FW_VERSION;
            st_msg.reset_reason = (uint8_t)esp_reset_reason();
            st_msg.fault_code = (uint8_t)(g_supervisor.latched_fault ? 1 : 0);
            st_msg.rx_err = parser.crc_errors;
            st_msg.ads_err = g_ads_err_count;
            st_msg.hb_age_ms = (uint16_t)hb_age;

            size_t enc_len = frame_encode(MSG_TYPE_STATUS, g_tx_seq++, (const uint8_t *)&st_msg, sizeof(st_msg), tx_buf, sizeof(tx_buf));
            if (enc_len > 0) Serial2.write(tx_buf, enc_len);
        }

        // 7. Periodic USB Monitor to Laptop (1 Hz)
        if (now - last_usb_debug_ms >= 1000) {
            last_usb_debug_ms = now;
            uint32_t hb_age = (g_supervisor.last_heartbeat_time_ms > 0) ? (now - g_supervisor.last_heartbeat_time_ms) : 9999;
            Serial.printf("[VAPA DIAG] Seq:%u | HB_Age:%lums | Armed:%d | OE:%d | EStop:%d | Fault:%d | ADS:%s | FSR:[%u,%u,%u,%u,%u]\n",
                          g_tx_seq, (unsigned long)hb_age,
                          g_supervisor.armed, g_supervisor.oe_enabled,
                          g_supervisor.estop_pressed, g_supervisor.latched_fault,
                          local_ads_ok ? "OK" : "ERR",
                          local_fsr[0], local_fsr[1], local_fsr[2], local_fsr[3], local_fsr[4]);
        }

        vTaskDelay(pdMS_TO_TICKS(2));
    }
}

// ==============================================================================
// 6. SETUP & LOOP
// ==============================================================================
void setup() {
    // 1. CRITICAL FAIL-CLOSED ACTION: Immediately disable PCA9685 OE
    pinMode(PIN_PCA9685_OE, OUTPUT);
    digitalWrite(PIN_PCA9685_OE, HIGH); // HIGH = all servo PWM lines disabled

    // 2. Hardware E-Stop pin (pullup)
    pinMode(PIN_ESTOP, INPUT_PULLUP);

    // 3. FSR ADC1 Configuration (0-3.3V range)
    analogSetAttenuation(ADC_11db);

    // 4. USB Debug Serial to Laptop
    Serial.begin(USB_BAUD_LAPTOP);
    delay(100);
    Serial.println("\n========================================================");
    Serial.println("  VAPA ESP32 Multi-Sensor & Supervisor (Rev 2 - Updated)");
    Serial.println("========================================================");
    Serial.println("  - TCA9548A & AS5600 Encoders: REMOVED");
    Serial.println("  - Protocol: High-Speed VAPA Binary Frame (460,800 baud)");
    Serial.println("  - Safety: Dual-core FreeRTOS supervisor with Watchdog");
    Serial.println("  - UART2: GPIO 17 (TX) -> Jetson Pin 10, GPIO 16 (RX) <- Pin 8");
    Serial.println("========================================================\n");

    // 5. I2C Initialization for ADS1115 (Fast mode: 400 kHz)
    Wire.begin(PIN_I2C_SDA, PIN_I2C_SCL, 400000);

    // 6. High-Speed UART2 to Jetson Orin Nano
    Serial2.begin(UART_BAUD_JETSON, SERIAL_8N1, PIN_UART_RX, PIN_UART_TX);

    // 7. Supervisor State & Mutex Initialization
    g_supervisor.init();
    g_data_mutex = xSemaphoreCreateMutex();

    // 8. Configure Hardware Task Watchdog (Core 2.x and 3.x compatible)
#if defined(ESP_IDF_VERSION_MAJOR) && (ESP_IDF_VERSION_MAJOR >= 5)
    esp_task_wdt_config_t twdt_config = {
        .timeout_ms = WATCHDOG_TIMEOUT_MS,
        .idle_core_mask = 0,
        .trigger_panic = true,
    };
    esp_task_wdt_init(&twdt_config);
#else
    esp_task_wdt_init(WATCHDOG_TIMEOUT_MS / 1000, true);
#endif

    // 9. Prime ADS1115 conversion
    ads1115_start_conversion(0);

    // 10. Launch FreeRTOS Tasks on Separate Cores
    xTaskCreatePinnedToCore(
        SensorSamplingTask,
        "SensorTask",
        4096,
        NULL,
        2,
        NULL,
        0 // Core 0: Sensor ADC sampling
    );

    xTaskCreatePinnedToCore(
        SupervisorUartTask,
        "SupervisorTask",
        4096,
        NULL,
        3,
        NULL,
        1 // Core 1: Supervisor State Machine & 460800 UART Link
    );
}

void loop() {
    // Execution handled by FreeRTOS tasks; idle loop
    vTaskDelay(pdMS_TO_TICKS(1000));
}
